Supabase Table Setup (SQL)

Run this SQL script in your Supabase project's SQL Editor tab to create the required books table and enable public access policies. 

-- 1. Create the books table
CREATE TABLE IF NOT EXISTS books (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  title TEXT NOT NULL,
  author TEXT,
  rating INT DEFAULT 5,
  genre TEXT,
  review TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Enable Row Level Security (RLS)
ALTER TABLE books ENABLE ROW LEVEL SECURITY;

-- 3. Create public access policy for easy demo CRUD operations
CREATE POLICY "Allow public access" 
ON books FOR ALL 
USING (true) 
WITH CHECK (true);
